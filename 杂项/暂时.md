func ResignerLogList(router *gin.Engine, container *nexus_wire.AppContainer) {
	api := router.Group("/Log-list") //注册一个路由组

	api.GET("/log", func(c *gin.Context) {
		log(c, container.GetDB())
	})
	api.GET("/log/detail", func(c *gin.Context) {
		logDetail(c, container.GetDB())
	})

}

// 查看列表
func log(c *gin.Context, db *gorm.DB) { //查看列表
	//解析查询条件
	var dto web_model.LogDataDTO
	if err := c.ShouldBind(&dto); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"参数解析失败\n": err.Error()})
		return
	}
	//调用Services层查询
	logs, err := service.QueryLogsByCondition(db, dto)
	if err != nil {
		c.JSON(http.StatusInternalServerError, gin.H{"查询错误\n": err.Error()})
		return
	}
	c.JSON(http.StatusOK, gin.H{
		"data": logs,
	})
}

package service

import (
	"ccm-nexus/pkg/sys_op_log_model"
	"ccm-nexus/pkg/web_model"
	"gorm.io/gorm"
	"strconv"
)

func QueryLogsByCondition(db *gorm.DB, dto web_model.LogDataDTO) ([]web_model.LogDataDTO, error) {
	var logs []web_model.LogDataDTO
	// 构造查询语句
	query := db.Model(&sys_op_log_model.SysOpLog{})
	// 如果某个字段的关键词不为空，则加入查询条件
	if dto.OpLogId != "" {
		query = query.Where("op_log_id = ?", dto.OpLogId)
	}
	if dto.OpPeopleName != "" {
		query = query.Where("op_people_name LIKE ?", "%"+dto.OpPeopleName+"%")
	}
	if dto.IpAddress != "" {
		query = query.Where("ip_address = ?", dto.IpAddress)
	}
	if dto.ModuleName != "" {
		query = query.Where("module_name LIKE ?", "%"+dto.ModuleName+"%")
	}
	if dto.OpType != "" {
		query = query.Where("op_type = ?", dto.OpType)
	}
	if dto.RequestModel != "" {
		query = query.Where("request_model = ?", dto.RequestModel)
	}
	if dto.OpStatus != "" {
		query = query.Where("op_status = ?", dto.OpStatus)
	}
	// 分页查询
	if dto.PageNumber != "" && dto.PageSize != "" {
		pageNumber, _ := strconv.Atoi(dto.PageNumber)
		pageSize, _ := strconv.Atoi(dto.PageSize)
		offset := (pageNumber - 1) * pageSize
		query = query.Offset(offset).Limit(pageSize)
	} else {
		query = query.Offset(0).Limit(10)
	}
	// 排序
	if dto.OrderBy != "" {
		query = query.Order("created_at " + dto.OrderBy)
	}
	// 执行查询
	if err := query.Find(&logs).Error; err != nil {
		return nil, err
	}
	return logs, nil

}
